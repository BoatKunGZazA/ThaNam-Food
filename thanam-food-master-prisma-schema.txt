// ============================================================
// THANAM FOOD — PRISMA DATABASE SCHEMA v1.0
// ============================================================
// Purpose:
//   Canonical Prisma data model for the Thanam Food restaurant
//   operating system.
//
// Architecture:
//   Modular Monolith + API
//   PostgreSQL = Source of Truth
//   Backend = Business Authority
//
// Rules:
//   - UUID primary keys
//   - Money = Decimal, never Float
//   - Time = DateTime with PostgreSQL timestamptz
//   - Historical business records are not physically deleted
//   - Order snapshots are immutable
//   - Payment status is independent from Order status
//   - Kitchen / Delivery / Printer are separate domains
//   - Audit Log is append-oriented
//   - Important mutations use Idempotency-Key / transactions
//
// NOTE:
//   PostGIS geometry fields require a PostgreSQL extension and
//   may require a custom SQL migration. The delivery-zone
//   boundary is therefore represented as Unsupported("geometry")
//   in Prisma.
//
// This schema is the database foundation. Business state-machine,
// authorization, validation, and transaction rules remain in the
// backend services and must NOT be inferred solely from Prisma.
// ============================================================

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

  ipAddress   String?  @db.VarChar(100)
  userAgent   String?
  requestId   String?  @db.VarChar(100)
  createdAt   DateTime @default(now()) @db.Timestamptz(6)

  store Store? @relation(fields: [storeId], references: [id], onDelete: SetNull)
  actor Staff? @relation(fields: [actorStaffId], references: [id], onDelete: SetNull)

  @@index([entityType, entityId, createdAt])
  @@index([actorStaffId, createdAt])
  @@index([storeId, createdAt])
  @@map("audit_logs")
}

model Setting {
  id        String   @id @default(uuid()) @db.Uuid
  storeId   String   @db.Uuid
  key       String   @db.VarChar(150)
  value     Json
  createdAt DateTime @default(now()) @db.Timestamptz(6)
  updatedAt DateTime @updatedAt @db.Timestamptz(6)

  store Store @relation(fields: [storeId], references: [id], onDelete: Restrict)

  @@unique([storeId, key])
  @@map("settings")
}

// ============================================================
// IMPORTANT IMPLEMENTATION NOTES
// ============================================================
//
// 1. PostGIS:
//    CREATE EXTENSION IF NOT EXISTS postgis;
//    DeliveryZone.boundary uses Unsupported("geometry").
//    Spatial GIST index should be created in SQL migration.
//
// 2. Partial unique indexes:
//    "Only one default address per customer" is a PostgreSQL
//    partial unique index and should be added through SQL:
//
//    CREATE UNIQUE INDEX ... ON customer_addresses(customer_id)
//    WHERE is_default = true AND status = 'ACTIVE';
//
// 3. Order state machine:
//    Prisma enum stores valid states but does NOT enforce valid
//    transitions. OrderService must enforce transitions.
//
// 4. Money:
//    Decimal(12,2) is authoritative. Never calculate business
//    money using JavaScript floating-point numbers.
//
// 5. Snapshots:
//    Order.customerSnapshot, Order.addressSnapshot and
//    OrderItem snapshots are historical records and must not be
//    rewritten when current customer/address/menu data changes.
//
// 6. Payment:
//    Payment.status and Order.paymentStatus are independent from
//    Order.status.
//
// 7. Audit:
//    Application code must write AuditLog records for critical
//    actions. Ordinary staff must not be able to edit/delete them.
//
// 8. Outbox:
//    Business transaction and OutboxEvent creation must occur in
//    the same database transaction.
//
// 9. Historical records:
//    Orders, payments, refunds, stock movements, timelines,
//    audit logs and other business history must not be physically
//    deleted through normal application operations.
//
// 10. Migration:
//     Development may use Prisma migrations.
//     Production must use:
//         prisma migrate deploy
//     Never use `prisma db push` as the production deployment
//     mechanism.
//
// 11. Seed:
//     Initial seed should create:
//       - one default store
//       - seven roles:
//         CUSTOMER, SUPER_ADMIN, OWNER, MANAGER, KITCHEN,
//         DELIVERY, ACCOUNTING
//       - baseline permissions
//       - one OWNER / SUPER_ADMIN bootstrap account through a
//         secure setup process, not a hard-coded production password.
//
// 12. Business validation not represented by DB alone:
//       - GPS range validation
//       - delivery polygon containment
//       - store opening/accepting/cutoff rules
//       - option min/max rules
//       - customer cancellation policy
//       - state transition permissions
//       - promotion stacking/usage race handling
//       - stock reconciliation
//       - printer retry policy
//
// ============================================================
// END OF SCHEMA v1.0
// ============================================================
